1. Analisis Koherensi SAR Sentinel-1
// WebGIS Gempa Flores – Analisis Koherensi SAR Sentinel-1 // Ganti ROI dengan batas wilayah Manggarai Timur yang sesungguhnya var roi = ee.Geometry.Rectangle([120.4, -8.9, 121.0, -8.4]); var tPra = ee.DateRange('2026-07-20', '2026-08-14'); var tPost = ee.DateRange('2026-08-15', '2026-08-30'); // Koleksi Sentinel-1 GRD var s1 = ee.ImageCollection('COPERNICUS/S1_GRD') .filter(ee.Filter.eq('instrumentMode', 'IW')) .filter(ee.Filter.listContains('transmitterReceiverPolarisation', 'VV')) .filterBounds(roi); var praMed = s1.filterDate(tPra).select('VV').median().clip(roi); var postMed = s1.filterDate(tPost).select('VV').median().clip(roi); // Hitung koherensi (proksi: normalisasi cross-correlation) var coherence = praMed.multiply(postMed) .divide(praMed.pow(2).add(postMed.pow(2)).sqrt()) .rename('coherence'); var deltaCoh = praMed.subtract(postMed).abs().rename('delta_coh'); Map.addLayer(deltaCoh, { min: 0, max: 0.6, palette: ['2f8f5b','e6c200','e08a00','b3261e'] }, 'Delta Koherensi SAR');
2. NDVI Change Detection (Landsat 8/9)
var ls = ee.ImageCollection('LANDSAT/LC09/C02/T1_L2') .filterBounds(roi) .filter(ee.Filter.lt('CLOUD_COVER', 30)); function maskCloud(img) { var qa = img.select('QA_PIXEL'); var mask = qa.bitwiseAnd(1 << 3).eq(0) .and(qa.bitwiseAnd(1 << 5).eq(0)); return img.updateMask(mask); } function addNDVI(img) { var ndvi = img.normalizedDifference(['SR_B5','SR_B4']).rename('NDVI'); return img.addBands(ndvi); } var ndviPra = ls.filterDate(tPra).map(maskCloud).map(addNDVI).select('NDVI').median(); var ndviPost = ls.filterDate(tPost).map(maskCloud).map(addNDVI).select('NDVI').median(); var deltaNDVI = ndviPost.subtract(ndviPra).rename('dNDVI'); Map.addLayer(deltaNDVI.clip(roi), { min: -0.4, max: 0.1, palette: ['b3261e','e08a00','ffffcc','2f8f5b'] }, 'ΔNDVI Landsat-9'); // Ekspor ke Google Drive Export.image.toDrive({ image: deltaNDVI.clip(roi), description: 'dNDVI_Flores_2026', scale: 30, region: roi, crs: 'EPSG:4326' });
